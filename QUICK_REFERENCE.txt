🚀 快速开始（需要 Node.js 环境）
═══════════════════════════════════════════════════════════════════

# 1️⃣ 验证所有优化
./scripts/verify-optimizations.sh

# 2️⃣ 安装依赖
npm install

# 3️⃣ 类型检查
npm run typecheck

# 4️⃣ 压力测试
node scripts/stress-test.js

# 5️⃣ 查看测试结果
cat stress-test-results/summary.txt

npm run build

npm run build:linux    # Linux
npm run build:mac      # macOS

# 或使用一键构建脚本
./scripts/build-release.sh           # Linux/macOS
scripts\build-release.bat            # Windows
